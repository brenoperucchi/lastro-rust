#!/bin/bash

# Function to create a backup tarball with a timestamp
create_backup() {
    local dir="$1"
    local timestamp=$(date +"%Y%m%d_%H%M%S")
    tar -czf "$dir-$timestamp.tar.gz" "$dir"
}

# Function to log backup creation
log_backup() {
    local timestamp="$1"
    echo "$(date '+%Y-%m-%d %H:%M:%S') - Backing up $1" >> backup_log.txt
}

# Main script execution
if [ $# -ne 1 ]; then
    echo "Usage: $0 <directory>"
    exit 1
fi

dir="$1"

# Check if the provided directory exists
if [ ! -d "$dir" ]; then
    echo "Error: Directory '$dir' does not exist."
    exit 2
fi

# Create the backup tarball with a timestamp
create_backup "$dir"

# Check if the backup was created successfully
if [ $? -ne 0 ]; then
    log_backup "$timestamp"
    exit 3
fi

# Check if the script was executed with the correct number of arguments
if [ "$#" -ne 2 ]; then
    log_backup "Usage: $0 <directory> <backup_timestamp>"
    exit 4
fi

# Extract the timestamp from the provided argument
timestamp=$(echo "$2" | cut -d'-' -f1)

# Check if the timestamp is valid
if ! [[ "$timestamp" =~ ^[0-9]{4}-[0-9]{2}-[0-9]{2} [0-9]{2}:[0-9]{2}:[0-9]{2}$ ]]; then
    log_backup "Invalid timestamp format. Expected: YYYY-MM-DD_HH:MM:SS"
    exit 5
fi

# Check if the backup directory exists
if [ ! -d "$dir-$timestamp" ]; then
    log_backup "Backup directory '$dir-$timestamp' does not exist."
    exit 6
fi

# Check if the backup directory is empty
if [ -z "$(ls -A "$dir-$timestamp")" ]; then
    log_backup "Backup directory '$dir-$timestamp' is empty."
    exit 7
fi

# Backup complete, exit with success
exit 0
