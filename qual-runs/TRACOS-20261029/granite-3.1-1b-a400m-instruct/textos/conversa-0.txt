```python
import csv
from typing import List, Tuple

def parse_csv(file_path: str) -> List[Tuple[str, str, str, int, float]]:
    """
    Parses a CSV file containing stock trade data.

    :param file_path: Path to the CSV file.
    :return: List of tuples containing date, symbol, side, quantity, and price.
    """
    trades = []
    with open(file_path, mode='r') as csvfile:
        reader = csv.DictReader(csvfile)
        for row in reader:
            date, symbol, side, quantity, price = row['date'], row['symbol'], row['side'], row['quantity'], row['price']
            trades.append((date, symbol, side, quantity, price))
    return trades


def calculate_daily_profit_loss(trades: List[Tuple[str, str, str, int, float]]) -> List[Tuple[str, str, str, int, float]]:
    """
    Computes daily profits and losses for each symbol.

    :param trades: List of tuples containing date, symbol, side, quantity, and price.
    :return: List of tuples containing date, symbol, side, quantity, and profit/loss.
    """
    daily_profits_losses = []
    for date, symbol, side, quantity, price in trades:
        if side == 'buy':
            daily_profit = price - price
        elif side == 'sell':
            daily_loss = price - price
        else:
            raise ValueError(f"Invalid side: {side}")

        daily_profits_losses.append((date, symbol, side, quantity, daily_profit, daily_loss))

    return daily_profits_losses


def main():
    file_path = 'stock_trades.csv'
    trades = parse_csv(file_path)
    daily_profits_losses = calculate_daily_profit_loss(trades)

    print("Daily Profits and Losses:")
    for date, symbol, side, quantity, daily_profit, daily_loss in daily_profits_losses:
        print(f"{date}: {symbol} - Side: {side}, Quantity: {quantity}, Profit/Loss: {daily_profit:.2f}, Loss: {daily_loss:.2f}")


if __name__ == "__main__":
    main()
```
