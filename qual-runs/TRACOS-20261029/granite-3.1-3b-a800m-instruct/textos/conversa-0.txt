```python
import pandas as pd

def calculate_daily_returns(csv_file: str) -> dict:
    """
    Reads a CSV file of stock trades and computes the daily profit and loss per symbol.

    :param csv_file: Path to the input CSV file
    :return: A dictionary containing daily returns for each symbol
    """
    try:
        # Read the CSV file
        df = pd.read_csv(csv_file)

        # Ensure required columns are present
        required_columns = {'date', 'symbol', 'side', 'quantity', 'price'}
        if not required_columns.issubset(df.columns):
            raise ValueError(f"Missing required columns: {required_columns - set(df.columns)}")

        # Calculate daily returns
        df['returns'] = df.groupby('symbol')['price'].pct_change()

        # Group by symbol and calculate daily returns
        daily_returns = df.groupby('symbol')['returns'].mean().reset_index()

        return daily_returns.to_dict(orient='records')

    except FileNotFoundError:
        raise FileNotFoundError(f"The file {csv_file} was not found.")
    except Exception as e:
        raise RuntimeError(f"An error occurred: {str(e)}")

# pytest tests
```
