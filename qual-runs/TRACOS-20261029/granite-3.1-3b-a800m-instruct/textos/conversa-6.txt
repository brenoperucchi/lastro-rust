```bash
#!/bin/bash

# Define variables
SOURCE_DIR="/path/to/your/directory"  # Replace with your source directory path
BACKUP_DIR="$HOME/backups"            # Default backup directory, can be overridden
TIMESTAMP=$(date +"%Y%m%d%H%M%S")       # Current timestamp
BACKUP_FILE="$BACKUP_DIR/backup_$(date +"%Y%m%d%H%M%S").tar.gz"  # Backup file name with timestamp
LOG_FILE="$BACKUP_DIR/backup.log"     # Log file path

# Check if source directory exists
if [ ! -d "$SOURCE_DIR" ]; then
  echo "Error: Source directory '$SOURCE_DIR' does not exist." >&2
  exit 1
fi

# Create backup directory if it doesn't exist
if [ ! -d "$BACKUP_DIR" ]; then
  mkdir -p "$BACKUP_DIR"
  if [ $? -ne 0 ]; then
    echo "Error: Failed to create backup directory '$BACKUP_DIR'." >&2
    exit 1
  fi
fi

# Create log file
touch "$LOG_FILE"
if [ $? -ne 0 ]; then
  echo "Error: Failed to create log file '$LOG_FILE'." >&2
  exit 1
fi

# Create tarball with timestamp and log to file
tar -czf "$BACKUP_FILE" -C "$SOURCE_DIR" . 2>> "$LOG_FILE"
if [ $? -ne 0 ]; then
  echo "Error: Failed to create backup '$BACKUP_FILE'." >&2
  exit 1
fi

echo "Backup created successfully: $BACKUP_FILE"
```
